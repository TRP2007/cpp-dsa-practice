/******************************************************************************
check if string is palindrome or not using recursion
*******************************************************************************/

#include <iostream>
using namespace std;

bool f(string s,int i)
{
    int n=s.length();
    if(i>=n/2)
    return true;
    if(s[i]!=s[n-i-1])
    return false;
    return f(s,i+1);
    
}
int main()
{
    string s;
    cin>>s;
    if (f(s,0))
    
        cout<<"palindrome";
    
    else
    
        cout<<"not palindrome";
    
    return 0;
}
